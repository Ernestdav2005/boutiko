  function esc(s) {
    return String(s == null ? "" : s).replace(/[&<>"']/g, function (c) {
      return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c];
    });
  }
  function fcfa(n) { return String(Math.round(Number(n) || 0)).replace(/\B(?=(\d{3})+(?!\d))/g, " "); }
  function $(s, el) { return (el || document).querySelector(s); }
  function toast(msg) {
    var old = document.querySelector(".toast"); if (old) old.remove();
    var t = document.createElement("div");
    t.className = "toast";
    t.setAttribute("role", "status");
    t.textContent = msg;
    document.body.appendChild(t);
    setTimeout(function () { t.remove(); }, 3200);
  }
  function api(path, method, body) {
    var opt = { method: method || "GET", headers: { "X-Requested-With": "boutiko" }, credentials: "same-origin" };
    if (body !== undefined) { opt.headers["Content-Type"] = "application/json"; opt.body = JSON.stringify(body); }
    return fetch(path, opt).then(function (r) {
      return r.json().catch(function () { return {}; }).then(function (d) {
        if (!r.ok) { var e = new Error(d.error || "Une erreur est survenue."); e.status = r.status; throw e; }
        return d;
      });
    });
  }
  function fmtDate(ts) {
    if (!ts) return "—";
    return new Date(ts * 1000).toLocaleDateString("fr-FR", { day: "2-digit", month: "2-digit", year: "numeric" });
  }

  // ---------- Chargement ----------
  function load() {
    Promise.all([api("/api/admin/overview"), api("/api/admin/shops"), api("/api/admin/branding")]).then(function (r) {
      O = r[0]; D = r[1]; BRAND = r[2]; draw();

      } else {

      }
    });

  function dueState(s) {
    var days = Math.ceil((s.paid_until * 1000 - Date.now()) / 86400000);
    if (s.paid_until && days < 0) return "expired";
    if (s.paid_until && days <= 3) return "soon";
    return "ok";
  }
  function isTrial(s) { return s.paid_until && s.paid_until <= s.created_at + (O.trialDays * 86400) + 60 && dueState(s) !== "expired"; }
  function relanceText(s) {
    var d = fmtDate(s.paid_until);
    var past = dueState(s) === "expired";
    return "Bonjour " + s.name + ", votre abonnement " + (past ? "est arrivé à échéance le " : "arrive à échéance le ") + d +
      ". Pour le renouveler, vous pouvez payer par Mobile Money et m'envoyer la confirmation ici. Merci !";
  }
  function revenue() {
    return D.shops.reduce(function (a, s) {
      if (s.status !== "active" || dueState(s) === "expired" || s.is_admin) return a;
      return a + (D.plans[s.plan] ? D.plans[s.plan].price : 0);
    }, 0);
  }

  // ---------- Boutiques : recherche, filtres, actions ----------
  function matches(s) {
    var st = dueState(s);
    if (F.plan !== "all" && s.plan !== F.plan) return false;
    if (F.status === "due" && !(s.status === "active" && st !== "ok" && !s.is_admin)) return false;
    if (F.status === "expired" && !(s.status === "active" && st === "expired" && !s.is_admin)) return false;
    if (F.status === "trial" && !(s.status === "active" && isTrial(s))) return false;
    if (F.status === "suspended" && s.status !== "suspended") return false;
    if (F.status === "blocked" && s.status !== "blocked") return false;
    if (F.q) {
      var q = F.q.toLowerCase();
      if ((s.name + " " + s.slug + " " + s.whatsapp).toLowerCase().indexOf(q) < 0) return false;
    }
    return true;
  }
  function viewShops(v) {

      }).join("") + "</div>" +

      Object.keys(D.plans).map(function (k) {
        return '<button class="btn small ' + (F.plan === k ? "" : "ghost") + '" data-p="' + k + '">' + esc(D.plans[k].label) + "</button>";
      }).join("") + "</div>";
    var list = D.shops.filter(matches);
    h += '<p class="muted small" id="cnt">' + list.length + (list.length > 1 ? " boutiques" : " boutique") + "</p><div id=\"list\">";

    h += list.map(shopCard).join("") + "</div>";
    v.innerHTML = h;

      var l = D.shops.filter(matches);

      $("#list", v).innerHTML = l.length ? l.map(shopCard).join("") : '<div class="empty"><b>Aucune boutique ici</b>Changez de filtre ou de recherche.</div>';
      bindShopActions(v);

    v.querySelectorAll("[data-f]").forEach(function (b) { b.onclick = function () { F.status = b.getAttribute("data-f"); draw(); }; });

    bindShopActions(v);
  }
  function shopCard(s) {
    var st = dueState(s);
    var badge;
    if (s.is_admin) badge = '<span class="chipx good">Propriétaire</span>';
    else if (s.status === "blocked") badge = '<span class="chipx blocked">Bloqué</span>';
    else if (s.status === "suspended") badge = '<span class="chipx zero">Suspendue</span>';
    else if (st === "expired") badge = '<span class="chipx zero">Expiré</span>';
    else if (st === "soon") badge = '<span class="chipx low">Bientôt échu</span>';
    else if (isTrial(s)) badge = '<span class="chipx trial">Essai</span>';
    else badge = '<span class="chipx good">À jour</span>';
    var plan = D.plans[s.plan] ? D.plans[s.plan].label : s.plan;
    var h = '<div class="sh' + (s.status === "blocked" ? " blocked" : "") + '" data-shop="' + s.id + '"><div class="top"><div><b><a href="/' + esc(s.slug) + '" target="_blank" rel="noopener">' + esc(s.name) + "</a></b>" +
      '<div class="muted small">+' + esc(s.whatsapp) + " · inscrit le " + fmtDate(s.created_at) + "</div></div>" + badge + "</div>" +
      '<div class="muted small" style="margin-top:6px">' + esc(plan) + " · " + s.pc + " produits · " + (s.bytes / 1048576).toFixed(1) + " Mo · " + s.v30 + " visites (30 j) · " + s.sold + " vente(s)" + (s.po ? " · " + s.po + " commande(s) en attente" : "") + "</div>" +
      '<div class="small" style="margin-top:4px">Abonnement jusqu\'au <b>' + fmtDate(s.paid_until) + "</b></div>" +
      '<div class="row"><select data-plan="' + s.id + '" style="width:auto;min-height:42px" aria-label="Formule">' + Object.keys(D.plans).map(function (k) {
        return '<option value="' + k + '"' + (k === s.plan ? " selected" : "") + ">" + D.plans[k].label + " (" + fcfa(D.plans[k].price) + ")</option>";
      }).join("") + "</select>" +
      '<button class="btn ok small" data-pay="' + s.id + '" data-d="30">Paiement reçu +30 jours</button>' +
      '<button class="btn ghost small" data-pay="' + s.id + '" data-d="90">+90 jours</button></div>' +
      '<div class="row"><a class="btn ghost small" target="_blank" rel="noopener" href="https://wa.me/' + esc(s.whatsapp) + "?text=" + encodeURIComponent(relanceText(s)) + '">Relancer sur WhatsApp</a>' +
      '<button class="btn ghost small" data-reset="' + s.id + '">Mot de passe temporaire</button></div>';
    if (!s.is_admin) {
      h += '<div class="row">' +
        (s.status === "active"
          ? '<button class="btn ghost small" data-sus="' + s.id + '" data-to="suspended">Suspendre</button>'
          : '<button class="btn ok small" data-sus="' + s.id + '" data-to="active">' + (s.status === "blocked" ? "Débloquer" : "Réactiver") + "</button>") +
        (s.status !== "blocked" ? '<button class="btn danger small" data-sus="' + s.id + '" data-to="blocked">Bloquer le compte</button>' : "") +
        "</div>";
    }
    return h + "</div>";
  }
  function refreshAfter(msg) {
    return function () { toast(msg); load(); };
  }
  function bindShopActions(v) {
    v.querySelectorAll("select[data-plan]").forEach(function (sel) {
      sel.onchange = function () {
        api("/api/admin/shops/" + sel.getAttribute("data-plan"), "POST", { plan: sel.value }).then(refreshAfter("Formule modifiée.")).catch(function (ex) { toast(ex.message); });
      };
    });
    v.querySelectorAll("[data-pay]").forEach(function (b) {
      b.onclick = function () {
        var d = parseInt(b.getAttribute("data-d"), 10) || 30;
        api("/api/admin/shops/" + b.getAttribute("data-pay"), "POST", { add_days: d }).then(refreshAfter("Abonnement prolongé de " + d + " jours.")).catch(function (ex) { toast(ex.message); });
      };
    });
    v.querySelectorAll("[data-sus]").forEach(function (b) {
      b.onclick = function () {
        var to = b.getAttribute("data-to");
        var q = to === "blocked" ? "Bloquer ce compte ? Le commerçant ne pourra plus se connecter et sa boutique sera cachée."
          : to === "suspended" ? "Suspendre cette boutique ? Ses clients ne pourront plus la voir." : "";
        if (q && !confirm(q)) return;
        var msg = to === "active" ? "Compte réactivé." : to === "blocked" ? "Compte bloqué." : "Boutique suspendue.";
        api("/api/admin/shops/" + b.getAttribute("data-sus"), "POST", { status: to }).then(refreshAfter(msg)).catch(function (ex) { toast(ex.message); });
      };
    });
    v.querySelectorAll("[data-reset]").forEach(function (b) {
      b.onclick = function () {
        if (!confirm("Créer un nouveau mot de passe temporaire pour ce commerçant ? L'ancien ne marchera plus.")) return;
        api("/api/admin/shops/" + b.getAttribute("data-reset") + "/reset", "POST", {}).then(function (d) {
          showPassword(d.password);
        }).catch(function (ex) { toast(ex.message); });
      };
    });
  }

    $("#pwc", ov).onclick = function () { if (navigator.clipboard) navigator.clipboard.writeText(pw).then(function () { toast("Mot de passe copié."); }); };
  }
