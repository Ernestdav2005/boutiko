  function compress(file, maxSide, maxBytes) {
    return new Promise(function (resolve, reject) {
      var url = URL.createObjectURL(file);
      var img = new Image();
      img.onload = function () {
        var s = Math.min(1, maxSide / Math.max(img.width, img.height));
        var w = Math.max(1, Math.round(img.width * s));
        var h = Math.max(1, Math.round(img.height * s));
        var c = document.createElement("canvas");
        c.width = w; c.height = h;
        var ctx = c.getContext("2d");
        ctx.fillStyle = "#fff";
        ctx.fillRect(0, 0, w, h);
        ctx.drawImage(img, 0, 0, w, h);
        URL.revokeObjectURL(url);
        var q = 0.74, tries = 0;
        (function go() {
          c.toBlob(function (blob) {
            if (!blob) return reject(new Error("Photo illisible."));
            if (blob.size > maxBytes && tries < 5) { q -= 0.1; tries++; return go(); }
            resolve(blob);
          }, "image/jpeg", q);
        })();
      };
      img.onerror = function () { URL.revokeObjectURL(url); reject(new Error("Photo illisible.")); };
      img.src = url;
    });
  }

  function ago(ts) {
    var m = Math.floor((Date.now() / 1000 - ts) / 60);
    if (m < 1) return "à l'instant";
    if (m < 60) return "il y a " + m + " min";
    var h = Math.floor(m / 60);
    if (h < 24) return "il y a " + h + " h";
    return "il y a " + Math.floor(h / 24) + " j";
  }

  var RC = { items: [], client: "", phone: "", discount: 0, record: true };
  function receiptText() {
    var s = S.me.shop;
    var total = receiptTotal();
    var lines = RC.items.map(function (i) { return "• " + i.name + " x" + i.qty + " : " + fcfa(i.price * i.qty) + " FCFA"; });
    var t = "🧾 Reçu — " + s.name + "\nDate : " + fmtDate(Date.now() / 1000) + (RC.client ? "\nClient : " + RC.client : "") + "\n\n" + lines.join("\n");
    if (RC.discount > 0) t += "\nRemise : -" + fcfa(RC.discount) + " FCFA";
    t += "\n\nTotal : " + fcfa(total) + " FCFA\nMerci pour votre achat !";
    return t;
  }
  function receiptTotal() {
    var t = RC.items.reduce(function (a, i) { return a + i.price * i.qty; }, 0);
    return Math.max(0, t - (Number(RC.discount) || 0));
  }

      var sel = $("#rsel").value, qty = Math.max(1, parseInt($("#rqty").value, 10) || 1);
      if (!sel) return toast("Choisissez un produit.");
      if (sel === "custom") {
        var n = $("#rcn").value.trim(), pr = parseInt($("#rcp").value, 10);
        if (!n || isNaN(pr) || pr < 0) return toast("Entrez le nom et le prix.");
        RC.items.push({ pid: null, name: n, price: pr, qty: qty });
        $("#rcn").value = ""; $("#rcp").value = "";
      } else {
        var p = S.products.filter(function (x) { return String(x.id) === sel; })[0];
        RC.items.push({ pid: p.id, name: p.name, price: p.promo_price !== null ? p.promo_price : p.price, qty: qty });
      }
      $("#rqty").value = 1;
      drawReceipt();
    };
    drawReceipt();
  }
  function drawReceipt() {
    var box = $("#rlist");
    if (!box) return;
    if (!RC.items.length) { box.innerHTML = '<p class="muted center" style="margin-top:14px">Ajoutez des articles pour calculer le total.</p>'; return; }
    box.innerHTML = RC.items.map(function (i, idx) {
      return '<div class="lrow"><div class="t"><b>' + esc(i.name) + " × " + i.qty + "</b><span>" + fcfa(i.price * i.qty) + ' FCFA</span></div><button class="btn danger small" data-rm="' + idx + '">Retirer</button></div>';
    }).join("") +
      '<div class="card"><label class="field"><span>Remise (FCFA, facultatif)</span><input id="rdisc" type="number" inputmode="numeric" min="0" value="' + (RC.discount || "") + '"></label>' +
      '<div class="total" id="rtotal"></div>' +
      '<label class="field"><span>Nom du client (facultatif)</span><input id="rcli" maxlength="60" value="' + esc(RC.client) + '"></label>' +
      '<label class="field"><span>WhatsApp du client (facultatif)</span><input id="rph" inputmode="tel" value="' + esc(RC.phone) + '"></label>' +
      '<label class="field" style="display:flex;gap:8px;align-items:center"><input type="checkbox" id="rrec" ' + (RC.record ? "checked" : "") + '> <span style="margin:0">Enregistrer la vente et retirer du stock</span></label>' +
      '<pre class="msg" id="rmsg"></pre>' +
      '<button class="btn block" id="rsend">Envoyer le reçu sur WhatsApp</button>' +
      '<button class="btn ghost block" id="rcopy" style="margin-top:8px">Copier le reçu</button>' +
      '<button class="btn danger block" id="rclear" style="margin-top:8px">Vider la vente</button></div>';
    function refresh() {
      RC.discount = Math.max(0, parseInt($("#rdisc").value, 10) || 0);
      RC.client = $("#rcli").value.trim(); RC.phone = $("#rph").value; RC.record = $("#rrec").checked;
      $("#rtotal").textContent = "Total : " + fcfa(receiptTotal()) + " FCFA";
      $("#rmsg").textContent = receiptText();
    }
    ["#rdisc", "#rcli", "#rph", "#rrec"].forEach(function (s) { $(s).oninput = refresh; $(s).onchange = refresh; });
    refresh();
    box.querySelectorAll("[data-rm]").forEach(function (b) {
      b.onclick = function () { RC.items.splice(parseInt(b.getAttribute("data-rm"), 10), 1); drawReceipt(); };
    });
    $("#rclear").onclick = function () { RC = { items: [], client: "", phone: "", discount: 0, record: true }; drawReceipt(); };
    $("#rcopy").onclick = function () {
      refresh();
      if (navigator.clipboard) navigator.clipboard.writeText(receiptText()).then(function () { toast("Reçu copié."); });
    };
    $("#rsend").onclick = function () {
      refresh();
      var text = receiptText();
      var phone = RC.phone ? normPhone(RC.phone) : "";
      var done = function () { window.open(waLink(phone, text), "_blank"); };
      if (!RC.record) return done();
      var items = RC.items.map(function (i) { return { product_id: i.pid, name: i.name, price: i.price, qty: i.qty }; });
      api("/api/sale", "POST", { items: items }).then(function () {
        toast("Vente enregistrée.");
        RC = { items: [], client: "", phone: "", discount: 0, record: true };
        return loadProducts();
      }).then(function () { done(); drawReceipt(); }).catch(function (ex) { toast(ex.message); });
    };
  }
  function normPhone(raw) {
    var d = String(raw || "").replace(/\D/g, "");
    if (d.indexOf("00") === 0) d = d.slice(2);
    if (d.indexOf("229") === 0) return d;
    if (d.length === 10 && d.indexOf("01") === 0) return "229" + d;
    if (d.length === 8) return "22901" + d;
    return d;
  }
