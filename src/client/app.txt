  function compress(file, maxSide, maxBytes) {
    return new Promise(function (resolve, reject) {
      var url = URL.createObjectURL(file);
      var img = new Image();
      img.onload = function () {
        var s = Math.min(1, maxSide / Math.max(img.width, img.height));
        var w = Math.max(1, Math.round(img.width * s));
        var h = Math.max(1, Math.round(img.height * s));
        var c = document.createElement("canvas");
        c.width = w; c.height = h;
        var ctx = c.getContext("2d");
        ctx.fillStyle = "#fff";
        ctx.fillRect(0, 0, w, h);
        ctx.drawImage(img, 0, 0, w, h);
        URL.revokeObjectURL(url);
        var q = 0.74, tries = 0;
        (function go() {
          c.toBlob(function (blob) {
            if (!blob) return reject(new Error("Photo illisible."));
            if (blob.size > maxBytes && tries < 5) { q -= 0.1; tries++; return go(); }
            resolve(blob);
          }, "image/jpeg", q);
        })();
      };
      img.onerror = function () { URL.revokeObjectURL(url); reject(new Error("Photo illisible.")); };
      img.src = url;
    });
  }

    return "il y a " + Math.floor(h / 24) + " j";
  }

  function normPhone(raw) {
    var d = String(raw || "").replace(/\D/g, "");

    if (d.length === 10 && d.indexOf("01") === 0) return "229" + d;

    return d;
  }

            return compress(file, 1200, 350 * 1024).then(function (blob) { return upload(blob, "gallery"); })
              .then(function (d) { galleryIds.push(d.id); drawGallery(); })
              .catch(function (ex) { var m = $("#galleryMsg"); if (m) m.textContent = ex.message; });
          });
        });
        chain.then(function () { var m = $("#galleryMsg"); if (m) m.textContent = "Photos prêtes. Appuyez sur « Enregistrer » pour les publier."; refreshProfilePreview(); });
      };
    }
    drawGallery();
    function refreshProfilePreview() {
      var preview = $("#profilePreview");
      if (!preview) return;
      var val = function (name) { var el = $("#sf [name=" + name + "]"); return el ? el.value.trim() : ""; };
      var brand = val("brand_name") || val("business_name") || "VOTRE MARQUE";
      var name = val("name") || "Nom de votre boutique";
      $("#profilePreviewBrand").textContent = brand;
      $("#profilePreviewName").textContent = name;
      $("#profilePreviewSlogan").textContent = val("slogan") || "Votre slogan apparaîtra ici.";
      $("#profilePreviewDescription").textContent = val("description") || "Présentez votre univers et vos produits.";
      preview.style.setProperty("--preview-color", color || "#0f766e");
      preview.setAttribute("data-theme", val("theme") || "editorial");
      var bannerId = pending.banner !== undefined ? pending.banner : s.banner_id;
      var cover = $("#profilePreviewCover");
      if (cover) cover.style.backgroundImage = bannerId && lim.custom ? 'url("/img/' + bannerId + '")' : "";
      var logoId = pending.logo !== undefined ? pending.logo : s.logo_id;
      var avatar = $("#profilePreviewAvatar");
      if (avatar) avatar.innerHTML = logoId ? '<img src="/img/' + logoId + '" alt="">' : esc((val("brand_name") || val("business_name") || name || "B").charAt(0).toUpperCase());
      var products = $("#profilePreviewProducts");
      if (products) {
        products.innerHTML = S.products.slice(0, 2).map(function (p) {
          var photo = p.photos && p.photos.length ? '<img src="/img/' + p.photos[0] + '" alt="">' : '<span class="profile-preview-product-art">✦</span>';
          return '<div class="profile-preview-product">' + photo + '<span>' + esc(p.name) + '</span><b>' + fcfa(p.promo_price || p.price) + ' FCFA</b></div>';
        }).join("") || '<div class="profile-preview-product profile-preview-placeholder"><span class="profile-preview-product-art">✦</span><span>Vos produits</span><b>apparaîtront ici</b></div><div class="profile-preview-product profile-preview-placeholder"><span class="profile-preview-product-art">✧</span><span>Vos belles photos</span><b>font la différence</b></div>';
      }
    }
    v.querySelectorAll('#sf [name="name"], #sf [name="business_name"], #sf [name="brand_name"], #sf [name="slogan"], #sf [name="description"], #sf [name="theme"]').forEach(function (el) {
      el.addEventListener("input", refreshProfilePreview);
      el.addEventListener("change", refreshProfilePreview);
    });
    refreshProfilePreview();
    $("#sf").onsubmit = function (e) {
      e.preventDefault();
      var f = formData(e.target), btn = $("#sf button[type=submit]");
      var chosen = []; for (var i = 0; i < 7; i++) if (f["d" + i]) chosen.push(i);
      var body = { name: f.name, business_name: f.business_name, brand_name: f.brand_name, slogan: f.slogan, description: f.description, whatsapp: f.whatsapp, color: color, theme: f.theme || "editorial", layout: f.layout || "grid", gallery: galleryIds, address: f.address, maps_url: f.maps_url, momo: f.momo,
        hours: f.open && f.close ? { open: f.open, close: f.close, days: chosen } : "" };
      if (pending.logo !== undefined) body.logo_id = pending.logo;
      if (lim.custom && pending.banner !== undefined) body.banner_id = pending.banner;
      btn.disabled = true;
      api("/api/shop", "PUT", body).then(function (d) { S.me.shop = d.shop; toast("Boutique enregistrée."); render(); })
        .catch(function (ex) { var er = $("#serr"); er.textContent = ex.message; er.style.display = "block"; btn.disabled = false; if (er.scrollIntoView) er.scrollIntoView({ behavior: "smooth", block: "center" }); });
    };
  }

  function viewMore(v) {
    v.innerHTML = '<p class="muted">Chargement…</p>';
    api("/api/stats").then(function (st) { drawMore(v, st); }).catch(function (ex) { if (ex.message !== "auth") v.innerHTML = '<div class="alert bad">' + esc(ex.message) + "</div>"; });
  }
  function drawMore(v, st) {
    var s = S.me.shop, m = S.me, lim = m.limits;
    var mb = (m.usage.bytes / 1048576).toFixed(1);
    var replies = (s.quick_replies || DEFAULT_REPLIES).split("\n").filter(function (x) { return x.trim(); });
    var admin = m.adminPhone;
    var h = "<h2>Mes chiffres</h2>" +
      '<div class="stat"><div class="card"><b>' + st.week.v + "</b>visites (7 jours)</div><div class=\"card\"><b>" + st.week.c + "</b>clics Commander (7 jours)</div>" +
      '<div class="card"><b>' + st.month.v + "</b>visites (30 jours)</div><div class=\"card\"><b>" + st.sales.n + "</b>ventes confirmées (30 jours)</div></div>" +
      '<div class="card" style="margin-top:8px"><b>' + fcfa(st.sales.total) + " FCFA</b> de ventes confirmées sur 30 jours</div>";
    if (st.top.length) h += '<div class="card" style="margin-top:8px"><b>Produits les plus demandés</b>' + st.top.map(function (t) { return "<div>" + esc(t.name) + " <span class=\"muted\">× " + t.q + "</span></div>"; }).join("") + "</div>";
    if (st.low.length) h += '<div class="alert warn" style="margin-top:8px"><b>Stock à renouveler</b>' + st.low.map(function (t) { return "<div>" + esc(t.name) + " : " + t.stock + " restant" + (t.stock > 1 ? "s" : "") + "</div>"; }).join("") + "</div>";
    h += '<h2 style="margin-top:18px">Réponses rapides</h2><p class="muted small">Copiez un message et collez-le dans WhatsApp. Une réponse par ligne, vous pouvez les modifier.</p>' +
      replies.map(function (r, i) { return '<div class="qrep"><p>' + esc(r) + '</p><button class="btn ghost small" data-copy="' + i + '">Copier</button></div>'; }).join("") +
      '<button class="btn ghost small" id="qedit" style="margin-top:6px">Modifier mes réponses</button>';
    h += '<h2 style="margin-top:18px">Mon abonnement</h2><div class="card"><p>Formule : <b>' + esc(lim.label) + "</b> (" + fcfa(lim.price) + " FCFA / mois)</p>" +
      "<p>Valable jusqu'au : <b>" + fmtDate(s.paid_until) + "</b></p><p class=\"muted small\">Produits : " + m.usage.products + " / " + lim.products + " · Photos : " + mb + " / " + lim.storageMb + " Mo</p>" +
      (admin ? '<a class="btn small" target="_blank" rel="noopener" href="' + esc(waLink(admin, "Bonjour, je suis " + s.name + ". Je voudrais renouveler ou changer ma formule.")) + '">Renouveler ou changer de formule</a>' : "") + "</div>";
    h += '<h2 style="margin-top:18px">Mon compte</h2><div class="card">' +
      '<button class="btn ghost small" id="mypw">Changer mon mot de passe</button> ' +
      '<a class="btn ghost small" href="/api/export">Télécharger mes données</a> ' +
      '<button class="btn danger small" id="logout">Me déconnecter</button></div>';
    v.innerHTML = h;
    v.querySelectorAll("[data-copy]").forEach(function (b) {
      b.onclick = function () { var t = replies[parseInt(b.getAttribute("data-copy"), 10)]; if (navigator.clipboard) navigator.clipboard.writeText(t).then(function () { toast("Message copié."); }); };
    });
    $("#qedit").onclick = function () {
      var txt = prompt("Une réponse par ligne :", replies.join("\n"));
      if (txt === null) return;
      api("/api/shop", "PUT", { quick_replies: txt }).then(function (d) { S.me.shop = d.shop; render(); }).catch(function (ex) { toast(ex.message); });
    };
    $("#mypw").onclick = openPw;
    $("#logout").onclick = function () { api("/api/logout", "POST", {}).then(function () { S.me = null; renderAuth(); }); };
  }
